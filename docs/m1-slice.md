```java
package com.euvcraft.lithmod.mc.data;

import com.euvcraft.lithmod.core.*;
import com.mojang.serialization.Codec;
import com.mojang.serialization.codecs.RecordCodecBuilder;
import java.util.List;
import java.util.Map;
import java.util.Optional;

/** Minecraft serialization adapter; the core record stays independent of game libraries. */
public final class MachineIngredientCodec {
    private MachineIngredientCodec() { }

    public static final Codec<MachineIngredient> CODEC = RecordCodecBuilder.create(instance -> instance.group(
        Codec.STRING.fieldOf("item").forGetter(ingredient -> ingredient.id()),
        Codec.INT.optionalFieldOf("count", 1).forGetter(MachineIngredient::count)
    ).apply(instance, MachineIngredient::item));

    public static final Codec<MachineIngredient> TAG_CODEC = RecordCodecBuilder.create(instance -> instance.group(
        Codec.STRING.fieldOf("tag").forGetter(ingredient -> ingredient.id()),
        Codec.INT.optionalFieldOf("count", 1).forGetter(MachineIngredient::count)
    ).apply(instance, MachineIngredient::tag));

    /**
     * Accepts either {@code {"item": ..., "count": ...}} or {@code {"tag": ..., "count": ...}},
     * so a datapack author picks the key that matches the shape of the ingredient. The item
     * form is tried first and falls through to the tag form when it does not apply.
     */
    public static final Codec<MachineIngredient> FLEXIBLE_CODEC =
        Codec.withAlternative(CODEC, TAG_CODEC);

}
```

#### src/main/java/com/euvcraft/lithmod/mc/data/MachineResultCodec.java

```java
package com.euvcraft.lithmod.mc.data;

import com.euvcraft.lithmod.core.*;
import com.mojang.serialization.Codec;
import com.mojang.serialization.codecs.RecordCodecBuilder;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import com.google.gson.JsonElement;
import com.google.gson.JsonParser;
import com.mojang.serialization.DataResult;
import com.mojang.serialization.Dynamic;
import com.mojang.serialization.JsonOps;

/** Minecraft serialization adapter; the core record stays independent of game libraries. */
public final class MachineResultCodec {
    private MachineResultCodec() { }

    /**
     * Reads one arbitrary JSON value and keeps it as compact text.
     *
     * <p>Written with explicit types because the inference through
     * {@code Codec.PASSTHROUGH.comapFlatMap} does not resolve on its own.</p>
     */
    private static final Codec<String> JSON_TEXT_CODEC =
        Codec.PASSTHROUGH.flatComapMap(
            (Dynamic<?> dynamic) -> dynamic.convert(JsonOps.INSTANCE).getValue().toString(),
            (String text) -> DataResult.success(
                new Dynamic<JsonElement>(JsonOps.INSTANCE, JsonParser.parseString(text))));

    public static final Codec<MachineResult> CODEC = RecordCodecBuilder.create(instance -> instance.group(
        Codec.STRING.fieldOf("item").forGetter(MachineResult::itemId),
        Codec.INT.optionalFieldOf("count", 1).forGetter(MachineResult::count),
        Codec.unboundedMap(Codec.STRING, JSON_TEXT_CODEC)
            .optionalFieldOf("components", Map.of()).forGetter(MachineResult::components)
    ).apply(instance, MachineResult::new));


}
```

MachineIngredient FLEXIBLE_CODEC accepts ("item", count) or ("tag", count); MachineResultCodec keeps result components as arbitrary JSON via a PASSTHROUGH codec (protocol object stays unquoted on disk).
